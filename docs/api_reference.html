Core Functions

createSDK(options)

Initialize the SDK - recommended entry point.

import { createSDK } from '@vikasbaplawat1/pulse-sdk'

const sdk = await createSDK({
  apiKey: process.env.PULSE_API_KEY,
  service: 'my-service',
  environment: 'production',
  debug: true,
})
Parameter Type Required Description
apiKey string Yes Project API key
service string No Service name (default: "unknown-service")
environment string No Environment (default: "production")

init(options)

Initialize the SDK - alternative to createSDK.

import { init } from '@vikasbaplawat1/pulse-sdk'
await init({ apiKey: '...', service: 'my-service' })

flush(), shutdown()

Manual flush and graceful shutdown.

import { flush, shutdown } from '@vikasbaplawat1/pulse-sdk'

// Flush pending events
await flush()

// Graceful shutdown with timeout
await shutdown(10000) // 10 second timeout

Capture Functions

captureError(error, options?)

Capture an exception or error.

import { captureError } from '@vikasbaplawat1/pulse-sdk'

try {
  await riskyOperation()
} catch (error) {
  captureError(error, {
    severity: 'error',
    context: { userId: '123' },
    fingerprint: 'custom-fingerprint'
  })
}

captureRequest(event)

Capture an HTTP request event.

import { captureRequest } from '@vikasbaplawat1/pulse-sdk'

captureRequest({
  type: 'request',
  requestId: 'req-123',
  url: '/api/users',
  method: 'GET',
  statusCode: 200,
  latency: 125,
  timestamp: Date.now(),
})

captureMessage(message, level?)

Capture a log message.

import { captureMessage } from '@vikasbaplawat1/pulse-sdk'

captureMessage('User logged in', 'info')

captureMetric(name, value, type, options?)

Capture a metric sample.

import { captureMetric } from '@vikasbaplawat1/pulse-sdk'

captureMetric('http.request.duration', 125.5, 'histogram', {
  tags: { method: 'GET' },
  unit: 'ms'
})

Scope Management

setUser(user | null)

Set the current user for all events.

import { setUser } from '@vikasbaplawat1/pulse-sdk'

setUser({ id: 'user-123', email: 'user@example.com' })
// Clear user
setUser(null)

setTag(key, value)

Set a tag for filtering.

import { setTag } from '@vikasbaplawat1/pulse-sdk'

setTag('environment', 'production')
setTag('version', '1.2.0')

setExtra(key, value)

Set extra context.

import { setExtra } from '@vikasbaplawat1/pulse-sdk'

setExtra('requestId', 'req-456')
setExtra('feature', 'new-dashboard')

setContext(name, context)

Set named context.

import { setContext } from '@vikasbaplawat1/pulse-sdk'

setContext('device', { model: 'iPhone 15', os: 'iOS 17' })
setContext('os', { name: 'macOS', version: '14'' })

Tracing

startSpan(name, options?)

Start a new span.

import { startSpan } from '@vikasbaplawat1/pulse-sdk'

const span = startSpan('db.query', {
  attributes: { 'db.system': 'postgres' }
})
span.setAttribute('db.statement', 'SELECT * FROM users')
// ... operation ...
span.end()

withSpan(name, fn, options?)

Execute function within a span.

import { withSpan } from '@vikasbaplawat1/pulse-sdk'

const result = await withSpan('process-file', async (span) => {
  span.setAttribute('file.size', file.size)
  return await processFile(file)
})

withSpanSync(name, fn, options?)

Synchronous version of withSpan.

import { withSpanSync } from '@vikasbaplawat1/pulse-sdk'

const result = withSpanSync('parse-json', (span) => {
  return JSON.parse(jsonString)
})

Context

addBreadcrumb(input)

Add a breadcrumb to the current context.

import { addBreadcrumb } from '@vikasbaplawat1/pulse-sdk'

addBreadcrumb({
  category: 'http',
  message: 'GET /api/users',
  type: 'http',
  data: { method: 'GET', statusCode: 200 }
})

getTraceId(), getSpanId()

Get current trace/span IDs.

import { getTraceId, getSpanId } from '@vikasbaplawat1/pulse-sdk'

const traceId = getTraceId()
const spanId = getSpanId()

Instrumentation

monitorMiddleware()

Express middleware for request tracking.

import { monitorMiddleware } from '@vikasbaplawat1/pulse-sdk'

const app = express()
app.use(monitorMiddleware())

instrumentHttpClient()

Auto-instrument outbound HTTP requests.

import { instrumentHttpClient } from '@vikasbaplawat1/pulse-sdk'

instrumentHttpClient() // Call once at startup
Note: All capture functions return void but are safe to call - errors are caught internally and never throw.
