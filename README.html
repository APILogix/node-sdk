Pulse APM Node.js SDK

An enterprise-grade observability and distributed tracing SDK built on OpenTelemetry with automated database interception, truthful timing attribution, synchronous ESM loader hooks, and first-class detached background trace support.

Turnkey Auto-Tracing

Automatic interception for PG, Redis, Mongo, MySQL, Prisma, HTTP clients, and frameworks with zero manual spans.

Truthful Timing Attribution

Zero synthetic clamping. Mathematical interval unions calculate genuine Node.js CPU vs database wait time.

Detached Background Traces

Post-response jobs (emails, webhooks) run as independent root traces connected by causal OTel SpanLink.

2. Installation & Module Preload

Install the package from npm:

npm install @pulse-apm/node
Crucial for Native ESM Applications ("type": "module")

In Node.js (>= 18.19.0 / >= 20.6.0), static import statements are resolved before your application code runs. To enable automatic interception for drivers like pg, redis, and ioredis, launch Node with the --import flag: 

node --import @pulse-apm/node/register server.js

For CommonJS projects (require), this flag is not needed; simply run node server.js.

3. SDK Initialization & Startup

Call init() at the very beginning of your application entry point before importing or creating database connection pools. apiKey is the only required option! The SDK automatically discovers your applicationName and release (version) directly from your project's package.json file. 

// server.js or app.ts (ESM)
import { init } from '@pulse-apm/node'

// 1. Initialize the SDK (apiKey is the only required parameter!)
await init({
  apiKey: process.env.PULSE_API_KEY || 'pk_live_your_api_key_here',
})

// 2. Import frameworks and database drivers AFTER initialization
import express from 'express'
import pg from 'pg'

const app = express()
// ... continue application setup
// server.js (CommonJS)
const { init } = require('@pulse-apm/node')

async function bootstrap() {
  // 1. Initialize SDK with apiKey only
  await init({
    apiKey: process.env.PULSE_API_KEY || 'pk_live_your_api_key_here',
  })

  // 2. Require dependencies AFTER initialization
  const express = require('express')
  const { Pool } = require('pg')

  const app = express()
  // ... continue application setup
}

bootstrap().catch(console.error)
ℹ
Process-Wide Telemetry Singleton

init() starts the process-level telemetry runtime, sets up OpenTelemetry providers, and registers global unhandled error handlers. All trace spans, metric counters, and middlewares automatically hook into this initialized process singleton.

4. Framework Integration: Express.js & Fastify

Mount monitorMiddleware() at the top of your middleware chain to capture request latency, body snapshots, route patterns, and HTTP headers. Mount expressErrorHandler() after all routes to capture unhandled exceptions with full trace correlation. 

import { init, monitorMiddleware, expressErrorHandler } from '@pulse-apm/node'
import express from 'express'
import pg from 'pg'

await init({ apiKey: process.env.PULSE_API_KEY! })

const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL })
const app = express()

app.use(express.json())

// Mount early: records request lifecycle & binds AsyncLocalStorage
app.use(monitorMiddleware())

// Standard route — database queries are automatically captured as child spans!
app.get('/api/users/:id', async (req, res) => {
  const { rows } = await pool.query('SELECT id, name, email FROM users WHERE id = $1', [req.params.id])
  if (rows.length === 0) return res.status(404).json({ error: 'Not found' })
  res.json({ user: rows[0] })
})

// Mount after all routes: correlates uncaught errors to the active trace
app.use(expressErrorHandler())

app.listen(3000, () => console.log('Server listening on port 3000'))
const { init, monitorMiddleware, expressErrorHandler } = require('@pulse-apm/node')
const express = require('express')
const { Pool } = require('pg')

async function start() {
  await init({ apiKey: process.env.PULSE_API_KEY })

  const pool = new Pool({ connectionString: process.env.DATABASE_URL })
  const app = express()

  app.use(express.json())
  app.use(monitorMiddleware())

  app.get('/api/orders/:id', async (req, res) => {
    const { rows } = await pool.query('SELECT * FROM orders WHERE id = $1', [req.params.id])
    res.json({ order: rows[0] })
  })

  app.use(expressErrorHandler())
  app.listen(3000, () => console.log('Server listening on port 3000'))
}

6. Custom Function & Business Logic Tracing (withSpan)

Use withSpan to measure specific functions, algorithms, or external service integrations within an active request: 

import { withSpan } from '@pulse-apm/node'

export async function processPayment(orderId: string, amount: number) {
  return await withSpan('payments.process', async (span) => {
    // 1. Enrich with searchable attributes
    span.setAttribute('payment.order_id', orderId)
    span.setAttribute('payment.amount', amount)
    span.setAttribute('payment.currency', 'USD')

    // 2. Execute business logic
    const gatewayResponse = await callPaymentGateway(orderId, amount)

    // 3. Authoritatively record status
    if (gatewayResponse.success) {
      span.setStatus('ok')
      span.setAttribute('payment.transaction_id', gatewayResponse.txnId)
      return gatewayResponse
    } else {
      span.setStatus('error', gatewayResponse.declineReason)
      throw new Error(`Payment declined: ${gatewayResponse.declineReason}`)
    }
  })
}
const { withSpan } = require('@pulse-apm/node')

async function calculateRiskScore(userProfile) {
  return await withSpan('fraud.calculate_risk', async (span) => {
    span.setAttribute('user.id', userProfile.id)
    span.setAttribute('user.country', userProfile.country)

    try {
      const score = await runFraudModel(userProfile)
      span.setAttribute('fraud.risk_score', score)
      span.setStatus('ok')
      return score
    } catch (err) {
      span.recordError(err)
      span.setStatus('error', err.message)
      throw err
    }
  })
}

7. Detached Background Work (withBackgroundSpan)

When an asynchronous operation outlives the HTTP response (e.g. fire-and-forget emails, webhooks, event ingestion), wrapping it with withBackgroundSpan prevents dropped spans and avoids memory leaks.

Why Not Use Regular withSpan For Background Jobs?
Standard withSpan (In-Request Only): Tied to the HTTP request trace. When the response finishes, the trace enters a 250ms quiescing deadline and finalizes. Any span executing after 250ms is permanently dropped to protect server memory.
withBackgroundSpan (Detached Execution): Spawns a new independent root trace with a fresh traceId and an isolated breadcrumb ring-buffer. Causally connects back to the originating HTTP trace using OpenTelemetry SpanLink ('caused_by'). 
import { withBackgroundSpan } from '@pulse-apm/node'
import express from 'express'

const app = express()

app.post('/api/register', async (req, res) => {
  // 1. Perform in-request database work
  const user = await db.createUser(req.body)

  // 2. Send response immediately (client doesn't wait for email)
  res.status(201).json({ success: true, userId: user.id })

  // 3. Detached background task with 10-second fail-open safety timeout
  void withBackgroundSpan('jobs.send_welcome_email', {
    timeoutMs: 10_000,
    attributes: {
      'user.id': user.id,
      'email.type': 'welcome_v2',
    }
  }, async (span) => {
    // Connected causally to the HTTP request via SpanLink!
    // Database and outgoing HTTP calls inside here are automatically traced
    await emailService.send(user.email, 'Welcome to Pulse!')
    await auditLog.record(user.id, 'email_dispatched')
    span.setStatus('ok')
  }).catch(err => {
    console.error('Background email failed:', err)
  })
})
const { withBackgroundSpan } = require('@pulse-apm/node')
const axios = require('axios')

function dispatchWebhook(eventPayload) {
  withBackgroundSpan('webhooks.dispatch', {
    timeoutMs: 15_000,
    attributes: {
      'webhook.event': eventPayload.type,
      'webhook.target_url': eventPayload.url,
    }
  }, async (span) => {
    const response = await axios.post(eventPayload.url, eventPayload.data, { timeout: 10000 })
    span.setAttribute('http.status_code', response.status)
    span.setStatus('ok')
  }).catch(err => {
    console.error('Webhook delivery failed:', err.message)
  })
}

module.exports = { dispatchWebhook }

8. Custom Database Spans (withDbSpan)

If you are using raw socket connections, internal database wrappers, or specialized databases (ClickHouse, Cassandra, SQLite): 

import { withDbSpan } from '@pulse-apm/node'

export async function queryAnalytics(sql: string) {
  return await withDbSpan('other', 'SELECT', async (span) => {
    span.setAttribute('db.name', 'analytics')
    span.setAttribute('db.sql.table', 'events')

    const result = await clickhouseClient.query(sql)
    return result
  }, {
    statement: sql, // Automatically sanitized / redacted
    captureStatement: true,
  })
}
const { withDbSpan } = require('@pulse-apm/node')

async function queryCassandra(queryStr, params) {
  return await withDbSpan('cassandra', 'SELECT', async (span) => {
    span.setAttribute('db.name', 'orders_keyspace')
    span.setAttribute('db.cassandra.consistency_level', 'LOCAL_QUORUM')

    const result = await cassandraClient.execute(queryStr, params)
    return result
  }, {
    statement: queryStr,
  })
}

module.exports = { queryCassandra }

9. Errors, Breadcrumbs & User Identity

Correlate diagnostic breadcrumbs, user profiles, and custom contextual tags with all outgoing errors and traces: 

import { captureError, addBreadcrumb, setUser, setTag, setExtra } from '@pulse-apm/node'

// 1. Identify user in auth middleware or handler
setUser({
  id: 'usr_9821',
  email: 'developer@company.com',
  tenantId: 'tenant_acme',
})

// 2. Add sequential breadcrumbs
addBreadcrumb({
  category: 'auth',
  message: 'User completed MFA challenge',
  level: 'info',
  data: { method: 'totp' },
})

setTag('region', 'us-east-1')
setExtra('active_feature_flags', ['checkout_v2', 'fast_pay'])

// 3. Capture caught exceptions with metadata
try {
  await executeOperation()
} catch (err) {
  captureError(err, {
    extra: { attempt: 3 },
    tags: { severity: 'critical' },
    fingerprint: ['gateway', 'timeout'],
  })
}
const { captureError, addBreadcrumb, setUser, setTag } = require('@pulse-apm/node')

setUser({ id: 'usr_102', email: 'user@test.com' })
addBreadcrumb({ category: 'checkout', message: 'User clicked pay' })
setTag('app_version', '8.1.0')

try {
  unstableCall()
} catch (err) {
  captureError(err)
}

10. Custom Metrics & Shutdown

Emit high-performance OpenTelemetry metrics, and flush telemetry safely during container termination: 

import { incrementCounter, recordHistogram, setGauge, shutdown } from '@pulse-apm/node'

// Counters
incrementCounter('orders.completed', 1, { plan: 'enterprise', region: 'us-east-1' })

// Histograms (latencies, payload sizes)
recordHistogram('payload.size_bytes', 48291, { route: '/api/upload' })

// Gauges (active connections, memory)
setGauge('worker.active_connections', 42)

// Clean container shutdown on SIGTERM / SIGINT
process.on('SIGTERM', async () => {
  console.log('Flushing telemetry before exit...')
  await shutdown(2000) // 2000ms maximum wait
  process.exit(0)
})
const { incrementCounter, shutdown } = require('@pulse-apm/node')
